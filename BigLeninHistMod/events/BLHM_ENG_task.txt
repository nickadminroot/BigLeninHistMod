add_namespace = BLHM_ENG_task

# FRA receives radar research bonus from joint radar development
country_event = {
	id = BLHM_ENG_task.1
	title = BLHM_ENG_task.1.t
	desc = BLHM_ENG_task.1.d
	picture = GFX_report_event_physics_lab_02

	is_triggered_only = yes

	option = {
		name = BLHM_ENG_task.1.a
		ai_chance = { factor = 100 }
		add_tech_bonus = {
			name = BLHM_ENG_joint_radar_bonus
			bonus = 1.0
			uses = 1
			category = radar_tech
		}
	}
}

# Monthly Soviet Aid - hidden deterministic check
# Fires every 30 days while SOV exists; gives equipment if SOV at war with GER
# Note: fighter_equipment is the generic archetype. If BBA DLC is active,
# individual airframe types (e.g. small_plane_airframe_1) would be more precise.
country_event = {
	id = BLHM_ENG_task.2
	title = BLHM_ENG_task.2.t
	desc = BLHM_ENG_task.2.d
	picture = GFX_report_event_generic_handshake

	is_triggered_only = yes
	hide_window = yes

	option = {
		name = BLHM_ENG_task.2.a
		ai_chance = { factor = 100 }
		if = {
			limit = { SOV = { exists = yes } }
			if = {
				limit = { SOV = { has_war_with = GER } }
				SOV = {
					add_equipment_to_stockpile = { type = fighter_equipment amount = 500 producer = ENG }
					add_equipment_to_stockpile = { type = motorized_equipment amount = 500 producer = ENG }
					add_equipment_to_stockpile = { type = anti_tank_equipment amount = 500 producer = ENG }
					add_equipment_to_stockpile = { type = anti_air_equipment amount = 500 producer = ENG }
				}
			}
			country_event = { id = BLHM_ENG_task.2 hours = 720 }
		}
	}
}
