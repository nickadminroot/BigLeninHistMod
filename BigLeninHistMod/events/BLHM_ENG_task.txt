# Soviet Economic Compromise — visible SOV event
# Accept: sets ENG flag, builds 2 civs in ENG capital, SOV gives oil rights, RAJ gives tungsten rights
# Refuse: sets ENG refusal flag, blocking further focuses
country_event = {
	id = BLHM_ENG_task.2
	title = BLHM_ENG_task.2.t
	desc = BLHM_ENG_task.2.d
	picture = GFX_report_event_generic_handshake

	is_triggered_only = yes

	option = {
		name = BLHM_ENG_task.2.a
		ai_chance = { factor = 80 }
		ENG = {
			set_country_flag = BLHM_ENG_soviet_compromise_accepted
			capital_scope = {
				add_extra_state_shared_building_slots = 2
				add_building_construction = {
					type = industrial_complex
					level = 2
					instant_build = yes
				}
			}
		}
		SOV = {
			limit = { exists = yes }
			give_resource_rights = {
				receiver = ENG
				state = 1083
				resources = { oil }
			}
		}
		RAJ = {
			limit = { exists = yes }
			give_resource_rights = {
				receiver = ROOT
				state = 288
				resources = { tungsten }
			}
			give_resource_rights = {
				receiver = ROOT
				state = 431
				resources = { tungsten }
			}
			give_resource_rights = {
				receiver = ROOT
				state = 434
				resources = { tungsten }
			}
		}
	}
	option = {
		name = BLHM_ENG_task.2.b
		ai_chance = { factor = 20 }
		ENG = {
			set_country_flag = BLHM_ENG_soviet_compromise_refused
		}
	}
}

# Soviet Lend-Lease — hidden deterministic monthly check
# Fires every 30 days while SOV exists and ENG has flag set
# Gives SOV equipment if SOV is at war with GER
# Uses ENG flag to avoid duplicate chain on re-fires
country_event = {
	id = BLHM_ENG_task.3
	title = BLHM_ENG_task.3.t
	desc = BLHM_ENG_task.3.d
	picture = GFX_report_event_generic_handshake

	is_triggered_only = yes
	hide_window = yes

	option = {
		name = BLHM_ENG_task.3.a
		ai_chance = { factor = 100 }
		if = {
			limit = {
				has_country_flag = BLHM_ENG_soviet_lend_lease_chain
				has_idea = BLHM_ENG_soviet_lend_lease_spirit
			}
			if = {
				limit = {
					SOV = { exists = yes }
					SOV = { has_war_with = GER }
				}
				SOV = {
					add_equipment_to_stockpile = { type = infantry_equipment amount = 10000 producer = ENG }
					add_equipment_to_stockpile = { type = support_equipment amount = 1000 producer = ENG }
					add_equipment_to_stockpile = { type = fighter_equipment amount = 500 producer = ENG }
					add_equipment_to_stockpile = { type = motorized_equipment amount = 500 producer = ENG }
					add_equipment_to_stockpile = { type = anti_tank_equipment amount = 500 producer = ENG }
					add_equipment_to_stockpile = { type = anti_air_equipment amount = 500 producer = ENG }
				}
			}
			country_event = { id = BLHM_ENG_task.3 hours = 720 }
		}
	}
}
