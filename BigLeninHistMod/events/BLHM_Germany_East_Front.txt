country_event = {
	id = blhm_germany_east_front.1
	hidden = yes
	is_triggered_only = yes

	trigger = {
		tag = GER
		has_war_with = SOV
		NOT = { has_country_flag = GER_east_front_war_with_ussr_started }
	}

	immediate = {
		set_country_flag = GER_east_front_war_with_ussr_started
		add_timed_idea = { idea = GER_east_front_war_with_ussr days = 710 }
		country_event = { id = blhm_germany_east_front.2 days = 710 }
	}
}

	hidden = yes
	is_triggered_only = yes

	trigger = {
		tag = GER
		has_country_flag = GER_east_front_war_with_ussr_started
		NOT = { has_country_flag = GER_east_front_blitzkrieg_timer_resolved }
	}

	immediate = {
		set_country_flag = GER_east_front_blitzkrieg_timer_resolved
		remove_ideas = GER_east_front_war_with_ussr

			limit = {
				219 = { is_controlled_by = GER }

		}

		}
	}
}
