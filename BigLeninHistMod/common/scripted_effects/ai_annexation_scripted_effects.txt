#####################################

# Germany - Update research slots based on puppeted nations
GER_update_research_slots_ai = {
	set_variable = { GER_ai_puppet_count = 0 }
	if = {
		limit = { has_country_flag = GER_romania_puppeted }
		change_variable = { GER_ai_puppet_count = 1 }
	}
	if = {
		limit = { has_country_flag = GER_bulgaria_puppeted }
		change_variable = { GER_ai_puppet_count = 1 }
	}
	if = {
		limit = { has_country_flag = GER_hungary_puppeted }
		change_variable = { GER_ai_puppet_count = 1 }
	}
	set_variable = { GER_ai_new_bonus = GER_ai_puppet_count }
	divide_variable = { GER_ai_new_bonus = 2 }
	set_temp_variable = { temp_bonus = GER_ai_new_bonus }
	add_to_temp_variable = { temp_bonus = 0.99 }
	round_temp_variable = temp_bonus
	set_temp_variable = { temp_current_bonus = 0 }
	if = {
		limit = { check_variable = { GER_ai_granted_bonus@0 > 0 } }
		set_temp_variable = { temp_current_bonus = GER_ai_granted_bonus@0 }
	}
	set_temp_variable = { temp_diff = temp_bonus }
	subtract_from_temp_variable = { temp_diff = temp_current_bonus }
	if = {
		limit = { check_variable = { temp_diff > 0 } }
		set_temp_variable = { temp_total = current_research_slots }
		add_to_temp_variable = { temp_total = temp_diff }
		set_research_slots = temp_total
		set_variable = { GER_ai_granted_bonus@0 = temp_bonus }
	}
}
