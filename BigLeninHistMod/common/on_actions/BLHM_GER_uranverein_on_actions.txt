on_actions = {
	on_state_control_changed = {
		effect = {
			# ROOT is the new controller, FROM the old controller, FROM.FROM the state.
			if = {
				limit = {
					FROM.FROM = { state = 110 }
					FROM = { tag = GER }
					GER = {
						has_completed_focus = GER_uranverein
						NOT = { controls_state = 110 }
						has_dynamic_modifier = GER_heavy_water_supplies_stable
					}
				}
				GER = { country_event = BLHM_GER_uranverein.1 }
			}

			if = {
				limit = {
					FROM.FROM = { state = 110 }
					tag = GER
					GER = {
						has_completed_focus = GER_uranverein
						controls_state = 110
						has_dynamic_modifier = GER_heavy_water_supplies_shortage
					}
				}
				GER = {
					remove_dynamic_modifier = { modifier = GER_heavy_water_supplies_shortage }
					add_dynamic_modifier = { modifier = GER_heavy_water_supplies_stable }
				}
			}
		}
	}
}
