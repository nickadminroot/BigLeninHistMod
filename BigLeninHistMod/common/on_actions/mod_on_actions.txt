	on_daily = {
		effect = { # Release the startup embargoes on 1940.1.1.
			if = {
				limit = {
					date > 1939.12.31
					date < 1940.1.2
				}
				every_country = {
					limit = {
						NOT = { tag = USA }
					}
					# Run both effects from their respective sending countries.
					break_embargo = USA
					USA = { break_embargo = PREV }
				}
				USA = {
					if = {
						limit = {
							has_idea = USA_zero_export_trade_law
						}
						swap_ideas = {
							remove_idea = USA_zero_export_trade_law
							add_idea = free_trade
						}
					}
				}
			}
		}

	on_monthly = {

			ROOT = { calculate_africa_control = yes }
			
			every_country = {
				if = {
					limit = {
						num_of_naval_factories > 30
					}
					set_variable = { fleet_expansion_penalty = num_of_naval_factories }

					multiply_variable = { fleet_expansion_penalty = 0.02 }
				}
				else = {

				}
