# Soviet Lend-Lease — recurring 30-day mission
# A delayed hidden event handles each activation so the mission is never
# restarted in the same tick in which it finishes.
BLHM_ENG_soviet_lend_lease = {
	BLHM_ENG_soviet_lend_lease_mission = {
		icon = GFX_goal_generic_occupy_states_coastal

		allowed = {
			original_tag = ENG
		}

		visible = {
			original_tag = ENG
			has_country_flag = BLHM_ENG_soviet_lend_lease_chain
			has_idea = BLHM_ENG_soviet_lend_lease_spirit
		}

		activation = { always = no }

		cancel_if_not_visible = no

		fire_only_once = no
		is_good = yes
		selectable_mission = no
		days_mission_timeout = 30

		timeout_effect = {
			if = {
				limit = {
					SOV = {
						exists = yes
						has_war_with = GER
					}
				}
				SOV = {
					add_equipment_to_stockpile = { type = infantry_equipment_1 amount = 10000 producer = ENG }
					add_equipment_to_stockpile = { type = support_equipment_1 amount = 1000 producer = ENG }
					add_equipment_to_stockpile = { type = fighter_equipment_1 amount = 500 producer = ENG }
					add_equipment_to_stockpile = { type = motorized_equipment_1 amount = 500 producer = ENG }
					add_equipment_to_stockpile = { type = anti_tank_equipment_1 amount = 500 producer = ENG }
					add_equipment_to_stockpile = { type = anti_air_equipment_1 amount = 500 producer = ENG }
				}
			}
			country_event = { id = BLHM_ENG_task.6 days = 1 }
		}
	}
}
