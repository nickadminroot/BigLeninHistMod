		original_tag = GER
	}
	
	visible = {
		original_tag = GER
	}
}

# Germany - Puppet Romania as Reichskommissariat
GER_annex_romania = {
	icon = generic_operation
	
	allowed = {
		original_tag = GER
	}
	
	available = {
		date > 1938.1.1
		ROM = {
			exists = yes
			is_subject = no
			is_ai = yes
		}
		has_government = fascism
	}
	
	fire_only_once = yes
	
	complete_effect = {
		ROM = {
			set_autonomy = {
				target = ROOT
				autonomy_state = autonomy_reichskommissariat
			}
			every_owned_state = {
				limit = { is_core_of = PREV }
				set_compliance = 100
			}
		}
		set_country_flag = GER_romania_puppeted
		custom_effect_tooltip = GER_annex_romania_tt
		hidden_effect = {
			GER_update_research_slots_ai = yes
		}
	}
	
	ai_will_do = {
		base = 100
		modifier = {
			factor = 0
			NOT = { is_ai = yes }
		}
	}
}
