			GER_update_research_slots_ai = yes
		}
	}
	
	ai_will_do = {
		base = 100
		modifier = {
			factor = 0
			NOT = { is_ai = yes }
		}
	}
}

# Britain - Annex India
ENG_annex_india = {
	icon = generic_operation
	
	allowed = {
		original_tag = ENG
	}
	
	available = {
		date > 1938.1.1
		RAJ = {
			exists = yes
			is_subject = yes
			is_ai = yes
		}
	}
	
	fire_only_once = yes
	
	complete_effect = {
		hidden_effect = {
			RAJ = {
				every_owned_state = {
					limit = { is_core_of = PREV }
					set_state_flag = BLHM_ai_annex_RAJ_core
				}
				# Remember the Soviet tungsten concessions before annexation removes them.
				every_owned_state = {
					limit = {
						OR = {
							state = 288 # Magwe
							state = 431 # West Bengal
							state = 434 # Arunachal Pradesh
						}
						has_resources_rights = {
							receiver = SOV
							resources = { tungsten }
						}
					}
					set_state_flag = BLHM_ai_annex_RAJ_soviet_tungsten
				}
			}
		}
		annex_country = { target = RAJ transfer_troops = yes }
		# Restore the concessions only after India has been annexed by Britain.
		every_owned_state = {
			limit = { has_state_flag = BLHM_ai_annex_RAJ_soviet_tungsten }
			ROOT = {
				give_resource_rights = {
					receiver = SOV
					state = PREV
					resources = { tungsten }
				}
			}
			clr_state_flag = BLHM_ai_annex_RAJ_soviet_tungsten
		}
		every_owned_state = {
			limit = { has_state_flag = BLHM_ai_annex_RAJ_core }
			set_compliance = 100
			clr_state_flag = BLHM_ai_annex_RAJ_core
		}
		set_country_flag = ENG_india_annexed
		hidden_effect = {
			ENG_update_research_slots_ai = yes
		}
	}
	
	ai_will_do = {
		base = 100
		modifier = {
			factor = 0
			NOT = { is_ai = yes }
		}
	}
}
}
